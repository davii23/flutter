
import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      home: Scaffold(
        appBar: AppBar(
          title: const Text('Responsive UI'),
          backgroundColor: Colors.blue,
        ),

        body: LayoutBuilder(
          builder: (context, constraints) {

            // Mobile Layout
            if (constraints.maxWidth < 600) {
              return Column(
                children: [
                  Card(
                    child: ListTile(
                      title: Text('Student'),
                    ),
                  ),
                  Card(
                    child: ListTile(
                      title: Text('Courses'),
                    ),
                  ),
                ],
              );
            }

            // Desktop / Tablet Layout
            return Row(
              children: [
                Expanded(
                  child: Card(
                    child: ListTile(
                      title: Text('Students'),
                    ),
                  ),
                ),
                Expanded(
                  child: Card(
                    child: ListTile(
                      title: Text('Courses'),
                    ),
                  ),
                ),
              ],
            );
          },
        ),
      ),
    );
  }
}
